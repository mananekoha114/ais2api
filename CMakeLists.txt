cmake_minimum_required(VERSION 3.25)
project(ais2api VERSION 0.1.0 LANGUAGES CXX)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(AIS2API_SANITIZE "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
include(CTest)
find_package(Boost CONFIG REQUIRED COMPONENTS json)
find_package(OpenSSL REQUIRED)
find_package(CURL REQUIRED)
find_package(SQLite3 REQUIRED)
if(TARGET SQLite3::SQLite3)
  set(AIS2API_SQLITE_TARGET SQLite3::SQLite3)
else()
  set(AIS2API_SQLITE_TARGET SQLite::SQLite3)
endif()
find_package(LibArchive REQUIRED)
find_package(Threads REQUIRED)
add_library(ais2api_core src/common.cpp src/config.cpp src/store.cpp src/pricing.cpp src/cookie_import.cpp src/schema.cpp src/protocol.cpp src/responses.cpp src/browser.cpp src/runtime.cpp src/setup.cpp src/download.cpp src/bundle.cpp src/server.cpp src/control.cpp src/management.cpp src/logs.cpp src/console.cpp src/log_sources.cpp src/webui.cpp src/region.cpp)
set(browser_bridge_inc "${CMAKE_CURRENT_BINARY_DIR}/generated/browser_bridge.inc")
target_sources(ais2api_core PRIVATE src/price_service.cpp src/usage.cpp)
add_custom_command(OUTPUT "${browser_bridge_inc}"
  COMMAND "${CMAKE_COMMAND}" "-DSOURCE=${CMAKE_CURRENT_SOURCE_DIR}/browser/bridge.js"
    "-DOUTPUT=${browser_bridge_inc}" -P "${CMAKE_CURRENT_SOURCE_DIR}/scripts/embed-browser.cmake"
  DEPENDS browser/bridge.js scripts/embed-browser.cmake VERBATIM)
target_sources(ais2api_core PRIVATE "${browser_bridge_inc}")
target_include_directories(ais2api_core PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
include(scripts/webui.cmake)
target_include_directories(ais2api_core PUBLIC include)
target_compile_definitions(ais2api_core PUBLIC BOOST_ASIO_NO_DEPRECATED BOOST_BEAST_USE_STD_STRING_VIEW)
target_compile_definitions(ais2api_core PRIVATE AIS2API_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(ais2api_core PUBLIC Boost::json Boost::headers OpenSSL::Crypto CURL::libcurl ${AIS2API_SQLITE_TARGET} LibArchive::LibArchive Threads::Threads)
if(WIN32)
  target_compile_definitions(ais2api_core PUBLIC _WIN32_WINNT=0x0A00 WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(ais2api_core PUBLIC ws2_32 mswsock advapi32 userenv bcrypt)
endif()
if(MSVC)
  set_property(TARGET ais2api_core PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(ais2api_core PUBLIC /utf-8 /W4 /permissive- /EHsc)
else()
  target_compile_options(ais2api_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(AIS2API_SANITIZE AND NOT MSVC)
  target_compile_options(ais2api_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(ais2api_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(ais2api src/main.cpp)
target_link_libraries(ais2api PRIVATE ais2api_core)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/userscript")
configure_file(userscript/ais2api-cookie.user.js userscript/ais2api-cookie.user.js COPYONLY)
if(MSVC)
  set_property(TARGET ais2api PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
if(BUILD_TESTING)
  find_package(Catch2 3 CONFIG REQUIRED)
  file(GLOB TEST_SOURCES CONFIGURE_DEPENDS tests/test_*.cpp)
  add_executable(ais2api_tests ${TEST_SOURCES})
  target_link_libraries(ais2api_tests PRIVATE ais2api_core Catch2::Catch2WithMain OpenSSL::SSL)
  if(MSVC)
    set_property(TARGET ais2api_tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  endif()
  add_test(NAME unit COMMAND ais2api_tests)
endif()
install(TARGETS ais2api RUNTIME DESTINATION bin)
install(FILES browser/bridge.js DESTINATION bin/browser)
install(FILES userscript/ais2api-cookie.user.js DESTINATION bin/userscript)
install(FILES runtime-manifest.json DESTINATION bin)
install(DIRECTORY schemas DESTINATION share/ais2api)
# One-click launchers at the archive root, beside bin/ and share/.
if(WIN32)
  install(FILES launcher/start.cmd DESTINATION .)
  install(FILES launcher/start-messages.txt DESTINATION share/ais2api)
else()
  install(PROGRAMS launcher/start.sh DESTINATION .)
endif()
include(scripts/browser-bundle.cmake)

# Optional developer fuzzing. This block does not change normal builds.
option(AIS2API_BUILD_FUZZER "Build the LLVM libFuzzer wire-parser harness" OFF)
if(AIS2API_BUILD_FUZZER)
  if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang" OR MSVC OR NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "AIS2API_BUILD_FUZZER requires desktop Linux Clang")
  endif()
  if(NOT AIS2API_SANITIZE)
    message(FATAL_ERROR "Enable AIS2API_SANITIZE with AIS2API_BUILD_FUZZER")
  endif()
  target_compile_options(ais2api_core PRIVATE -fsanitize=fuzzer-no-link)
  add_executable(ais2api_fuzz_wire tests/fuzz_wire.cpp)
  target_link_libraries(ais2api_fuzz_wire PRIVATE ais2api_core)
  target_compile_options(ais2api_fuzz_wire PRIVATE -fsanitize=fuzzer,address,undefined)
  target_link_options(ais2api_fuzz_wire PRIVATE -fsanitize=fuzzer,address,undefined)
endif()

# Runtime archives use the same explicitly installed files as `cmake --install`.
install(FILES LICENSE docs/platforms.md DESTINATION share/doc/ais2api)
install(FILES NOTICE THIRD_PARTY_NOTICES.md docs/github-publishing.md DESTINATION share/doc/ais2api)
install(FILES licenses/CloakBrowser-BINARY-LICENSE.md DESTINATION share/doc/ais2api/browser)
install(FILES docs/README.md docs/getting-started.md docs/clients.md docs/configuration.md docs/troubleshooting.md
              docs/cli.md docs/protocol.md docs/responses.md docs/true-role.md docs/browser.md
              docs/validation.md docs/performance.md docs/roadmap.md docs/security-review.md
              docs/userscript.md docs/management-api.md docs/webui.md docs/docker.md DESTINATION share/doc/ais2api)
install(DIRECTORY docs/reports DESTINATION share/doc/ais2api)
if(DEFINED VCPKG_INSTALLED_DIR AND DEFINED VCPKG_TARGET_TRIPLET)
  install(DIRECTORY "${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/share/"
    DESTINATION share/licenses/ais2api-dependencies
    FILES_MATCHING PATTERN copyright)
endif()
set(CPACK_PACKAGE_NAME "ais2api")
set(CPACK_PACKAGE_VENDOR "Mana Nekoha (mananekoha114)")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Native local AI Studio API gateway")
set(CPACK_PACKAGE_VERSION "${PROJECT_VERSION}")
set(CPACK_RESOURCE_FILE_LICENSE "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE")
set(CPACK_PACKAGE_FILE_NAME "ais2api-${PROJECT_VERSION}-${CMAKE_SYSTEM_NAME}-${CMAKE_SYSTEM_PROCESSOR}")
if(AIS2API_BROWSER_BUNDLE_DIR)
  string(APPEND CPACK_PACKAGE_FILE_NAME "-Browser-Local")
endif()
set(CPACK_PACKAGE_DIRECTORY "${CMAKE_BINARY_DIR}/packages")
set(CPACK_PACKAGE_CHECKSUM SHA256)
set(CPACK_MONOLITHIC_INSTALL ON)
set(CPACK_INCLUDE_TOPLEVEL_DIRECTORY ON)
set(CPACK_VERBATIM_VARIABLES ON)
if(WIN32)
  set(CPACK_GENERATOR "ZIP")
else()
  set(CPACK_GENERATOR "TGZ")
endif()
include(CPack)
# CPack otherwise adds a whole-working-tree source archive target. Ship only the
# explicit runtime install list; never accidentally package local account files.
file(REMOVE "${CMAKE_CURRENT_BINARY_DIR}/CPackSourceConfig.cmake")

# Developer-only offline audit tools; excluded from install/package rules.
option(AIS2API_BUILD_PROTOCOL_PROBE "Build the offline request normalization probe" OFF)
if(AIS2API_BUILD_PROTOCOL_PROBE)
  add_executable(ais2api_protocol_probe tests/protocol_probe.cpp)
  target_link_libraries(ais2api_protocol_probe PRIVATE ais2api_core)
  if(MSVC)
    set_property(TARGET ais2api_protocol_probe PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  endif()
endif()

option(AIS2API_BUILD_REPLAY "Build the synthetic loopback HTTP replay tool" OFF)
if(AIS2API_BUILD_REPLAY)
  add_executable(ais2api_replay tools/replay.cpp)
  target_link_libraries(ais2api_replay PRIVATE ais2api_core)
  if(WIN32)
    target_link_libraries(ais2api_replay PRIVATE psapi)
  endif()
endif()
