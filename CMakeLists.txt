cmake_minimum_required(VERSION 3.25)
# The experimental in-process backend links a SpiderMonkey built outside this
# tree (tools/spidermonkey). Off, nothing of it is compiled or downloaded.
option(AIS2API_SPIDERMONKEY "Build the experimental SpiderMonkey backend" OFF)
set(AIS2API_SPIDERMONKEY_DIR "" CACHE PATH "The artifact tools/spidermonkey builds: lib/ and BUILDINFO.json")
if(AIS2API_SPIDERMONKEY)
  list(APPEND VCPKG_MANIFEST_FEATURES spidermonkey)
endif()
# The SpiderMonkey backend's network layer: curl-impersonate's libcurl, set to
# Firefox's TLS and HTTP/2 fingerprints (scripts/curl-impersonate.cmake). It
# replaces the program's libcurl. On its own, it lets the network layer be
# built and tested without SpiderMonkey.
option(AIS2API_CURL_IMPERSONATE "Use curl-impersonate's libcurl, with Firefox's network fingerprints" ${AIS2API_SPIDERMONKEY})
if(AIS2API_CURL_IMPERSONATE)
  # vcpkg's libcurl is the default feature; its headers would shadow these.
  set(VCPKG_MANIFEST_NO_DEFAULT_FEATURES ON)
  list(APPEND VCPKG_MANIFEST_FEATURES curl-impersonate)
endif()
project(ais2api VERSION 0.1.0 LANGUAGES CXX)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

option(AIS2API_SANITIZE "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)

# carry Camoufox. Release archives never come from one.

  set(AIS2API_ZSTD_TARGET zstd::libzstd_static)

# What every library here builds on: errors, JSON and file helpers.
add_library(ais2api_common STATIC src/common.cpp)

target_link_libraries(ais2api_common PUBLIC Boost::json Boost::headers OpenSSL::Crypto Threads::Threads)
if(WIN32)
  target_compile_definitions(ais2api_common PUBLIC _WIN32_WINNT=0x0A00 WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(ais2api_common PUBLIC advapi32)
endif()
if(MSVC)
  target_compile_options(ais2api_common PUBLIC /utf-8 /W4 /permissive- /EHsc)
else()
  target_compile_options(ais2api_common PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(AIS2API_SANITIZE AND NOT MSVC)
  target_compile_options(ais2api_common PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(ais2api_common PUBLIC -fsanitize=address,undefined)
endif()

  target_sources(ais2api_core PRIVATE src/spidermonkey.cpp)
endif()
add_custom_command(OUTPUT "${browser_bridge_inc}"
  COMMAND "${CMAKE_COMMAND}" "-DSOURCE=${CMAKE_CURRENT_SOURCE_DIR}/browser/bridge.js"

    "-DOUTPUT=${camoufox_catalog_inc}" -P "${CMAKE_CURRENT_SOURCE_DIR}/scripts/embed-camoufox.cmake"
  DEPENDS ${camoufox_catalog_files} scripts/embed-camoufox.cmake VERBATIM)
target_sources(ais2api_core PRIVATE "${camoufox_catalog_inc}")
if(AIS2API_DEV_BUNDLE)
  set(AIS2API_FPGEN_MODEL_DIR "" CACHE PATH "The fpgen model files runtime-manifest.json pins, embedded in a dev bundle")
  if(NOT AIS2API_FPGEN_MODEL_DIR)
    message(FATAL_ERROR "A dev bundle embeds the fpgen model: set AIS2API_FPGEN_MODEL_DIR (scripts/build-dev.ps1 does)")
  endif()
  get_filename_component(fpgen_model_dir "${AIS2API_FPGEN_MODEL_DIR}" ABSOLUTE)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS runtime-manifest.json)
  file(READ runtime-manifest.json runtime_manifest)
  string(JSON fpgen_file_count LENGTH "${runtime_manifest}" fpgen_model file_sha256)
  math(EXPR fpgen_last "${fpgen_file_count} - 1")
  set(fpgen_model_files "")
  foreach(index RANGE ${fpgen_last})
    string(JSON name MEMBER "${runtime_manifest}" fpgen_model file_sha256 ${index})
    list(APPEND fpgen_model_files "${fpgen_model_dir}/${name}")
  endforeach()
  set(fpgen_model_inc "${CMAKE_CURRENT_BINARY_DIR}/generated/fpgen_model.inc")
  add_custom_command(OUTPUT "${fpgen_model_inc}"
    COMMAND "${CMAKE_COMMAND}" "-DMODEL_DIR=${fpgen_model_dir}"
      "-DMANIFEST=${CMAKE_CURRENT_SOURCE_DIR}/runtime-manifest.json"
      "-DOUTPUT=${fpgen_model_inc}" -P "${CMAKE_CURRENT_SOURCE_DIR}/scripts/embed-fpgen.cmake"
    DEPENDS ${fpgen_model_files} runtime-manifest.json scripts/embed-fpgen.cmake VERBATIM)
  target_sources(ais2api_core PRIVATE "${fpgen_model_inc}")
endif()

  target_compile_definitions(ais2api_tests PRIVATE AIS2API_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
  if(MSVC)
    set_property(TARGET ais2api_tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  endif()
  if(AIS2API_CURL_IMPERSONATE)
    ais2api_copy_curl_impersonate(ais2api_tests)
    # The fingerprint tests decode the HTTP/2 headers the client sends.
    find_path(AIS2API_NGHTTP2_INCLUDE_DIR nghttp2/nghttp2.h REQUIRED)
    find_library(AIS2API_NGHTTP2_LIBRARY NAMES nghttp2 REQUIRED)
    target_include_directories(ais2api_tests PRIVATE "${AIS2API_NGHTTP2_INCLUDE_DIR}")
    target_link_libraries(ais2api_tests PRIVATE "${AIS2API_NGHTTP2_LIBRARY}")
    target_compile_definitions(ais2api_tests PRIVATE NGHTTP2_STATICLIB)
  endif()
  add_test(NAME unit COMMAND ais2api_tests)
  # ais2api_common, ais2api_bidi and ais2api_waa stay free of the service.
  add_test(NAME layering COMMAND "${CMAKE_COMMAND}" "-DSOURCE_DIR=${CMAKE_CURRENT_SOURCE_DIR}"
    -P "${CMAKE_CURRENT_SOURCE_DIR}/scripts/check-layering.cmake")

  if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang" OR MSVC OR NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "AIS2API_BUILD_FUZZER requires desktop Linux Clang")

    message(FATAL_ERROR "Enable AIS2API_SANITIZE with AIS2API_BUILD_FUZZER")
